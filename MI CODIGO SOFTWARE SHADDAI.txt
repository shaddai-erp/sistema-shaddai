from fpdf import FPDF

def imprimir_factura_sobre_formato(nombre_cliente, rif_cliente, direccion_cliente, productos, total):
    # Creamos un PDF transparente (sin líneas, solo los textos)
    pdf = FPDF(orientation='P', unit='mm', format='Letter')
    pdf.add_page()
    pdf.set_font("Arial", size=11)
    
    # --- AQUÍ DEFINIMOS LAS COORDENADAS (X, Y) ---
    # X es la distancia desde la izquierda, Y es la distancia desde arriba
    pdf.text(35, 65, f"{nombre_cliente}")   # Nombre o Razón Social
    pdf.text(35, 72, f"{rif_cliente}")      # No. de RIF
    pdf.text(120, 72, "0412-1234567")       # Teléfono (ejemplo)
    
    # Imprimir productos
    y_pos = 100 
    for item in productos:
        pdf.text(35, y_pos, item['desc'])   # Producto
        pdf.text(140, y_pos, str(item['cant'])) # Cantidad
        y_pos += 7 # Salto de línea
        
    pdf.text(170, 250, f"{total}")          # Total a pagar
    
    pdf.output("factura_lista.pdf")

def conectar_db():
    conn = sqlite3.connect("mi_negocio.db")
    cursor = conn.cursor()
    
    # 1. Tabla Inventario
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS inventario (
            codigo TEXT,
            descripcion TEXT,
            existencia REAL,
            costo REAL,
            precio REAL,
            sucursal TEXT,
            clasificacion TEXT,
            unidad TEXT,
            stock_minimo REAL DEFAULT 0,
            PRIMARY KEY (codigo, sucursal)
        )''')
    
    # 2. Tabla Categorías
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS categorias (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            nombre TEXT UNIQUE
        )''')
    
    # 3. Tabla Movimientos
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS movimientos (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            fecha TEXT,
            codigo TEXT,
            tipo_movimiento TEXT,
            cantidad REAL,
            sucursal TEXT
        )''')
    
    # 4. Tabla de Proveedores
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS proveedores (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            rif TEXT UNIQUE,
            razon_social TEXT,
            direccion TEXT,
            telefono TEXT
        )''')
    
    # 5. Tabla de Compras
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS compras (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            tipo_doc TEXT,
            num_documento TEXT,
            num_control TEXT,
            proveedor_rif TEXT,
            proveedor_nombre TEXT,
            fecha_compra TEXT,
            monto_total REAL,
            base_imponible REAL DEFAULT 0,
            monto_iva REAL DEFAULT 0,
            retencion_applied REAL DEFAULT 0,
            tipo_pago TEXT,
            estado TEXT
        )''')

    # 6. Tabla de Cuentas por Pagar (CxP)
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS cuentas_por_pagar (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            id_compra INTEGER,
            proveedor TEXT,
            num_factura TEXT,
            monto_original REAL,
            saldo_pendiente REAL,
            fecha_vencimiento TEXT,
            estado TEXT,
            FOREIGN KEY(id_compra) REFERENCES compras(id)
        )''')

    # 7. Tabla de Carrito Temporal de Compras
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS carrito_temporal (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            codigo TEXT,
            descripcion TEXT,
            cantidad REAL,
            costo_unitario REAL,
            subtotal REAL,
            es_nuevo INTEGER,
            clasificacion TEXT,
            unidad TEXT,
            precio_venta REAL
        )''')

    # 8. Tabla de Clientes
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS clientes (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            cedula_rif TEXT UNIQUE,
            razon_social TEXT,
            telefono TEXT,
            direccion TEXT
        )''')

    # 9. Tabla de Ventas
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS ventas (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            fecha_venta TEXT,
            cliente_doc TEXT,
            cliente_nombre TEXT,
            monto_bruto REAL,
            descuento_aplicado REAL,
            monto_final_neto REAL,
            condicion_pago TEXT,
            metodo_pago TEXT,
            estado TEXT
        )''')

    # 10. Tabla de Carrito Temporal de Ventas
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS carrito_ventas (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            codigo TEXT,
            descripcion TEXT,
            cantidad REAL,
            precio_lista REAL,
            subtotal REAL
        )''')
        
    conn.commit()
    return conn
